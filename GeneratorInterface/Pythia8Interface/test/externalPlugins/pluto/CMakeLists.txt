cmake_minimum_required(VERSION 3.15)
project(CMSPlutoPlugin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(ROOT REQUIRED COMPONENTS Core Physics EG)
find_path(PLUTO_INCLUDE PChannel.h HINTS "${PLUTO_ROOT}/include" REQUIRED)
find_library(PLUTO_LIBRARY Pluto HINTS "${PLUTO_ROOT}/lib" "${PLUTO_ROOT}/lib64" REQUIRED)
find_path(PYTHIA_INCLUDE Pythia8/Pythia.h HINTS "${PYTHIA_ROOT}/include" REQUIRED)
find_library(PYTHIA_LIBRARY pythia8 HINTS "${PYTHIA_ROOT}/lib" REQUIRED)
add_library(CMSPluto SHARED PlutoPlugin.cc)
target_include_directories(CMSPluto PRIVATE "${PLUTO_INCLUDE}" "${PYTHIA_INCLUDE}")
target_link_libraries(CMSPluto PRIVATE "${PLUTO_LIBRARY}" "${PYTHIA_LIBRARY}" ROOT::Core ROOT::Physics ROOT::EG)
install(TARGETS CMSPluto LIBRARY DESTINATION lib)
